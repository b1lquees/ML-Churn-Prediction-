# Runtime dependencies for SERVING (this is what the Docker image installs).
#
# Deliberately small: mlflow is NOT here. Training logs the model with MLflow,
# but serving loads the pickle inside the run directory directly, so the runtime
# skips mlflow and everything it pulls in (sqlalchemy, alembic, flask, gunicorn,
# pyarrow, docker). That is what keeps the image small enough for a free-tier
# instance. Training deps live in requirements-dev.txt.
#
# Versions of numpy / pandas / scikit-learn / xgboost must stay compatible with
# the pickled model in src/serving/model/, so they are pinned to the versions
# recorded in that model's own requirements.txt. Do not bump one without
# retraining the model.

# --- web layer ---
fastapi==0.115.0
uvicorn==0.30.5
starlette==0.38.6
pydantic==2.8.2
pydantic_core==2.20.1

# REQUIRED, not optional: src/app/main.py does `import gradio as gr` at module
# import time and mounts the UI at /ui. Removing gradio here does not slim the
# runtime, it makes the container die on startup with ModuleNotFoundError,
# because the Docker image installs only this file.
gradio==4.44.1
# Pinned together with gradio, never separately. gradio 4.x imports HfFolder,
# which huggingface_hub removed in 1.0, so an unpinned huggingface_hub resolves
# to 1.x and every fresh install fails at `import gradio` with
# "cannot import name 'HfFolder'".
huggingface_hub>=0.25,<1.0

# --- model loading + inference ---
numpy==1.26.4
pandas==2.1.4
scikit-learn==1.5.2
scipy==1.16.1
xgboost==3.0.3
joblib==1.4.2
# The estimator was pickled with cloudpickle by MLflow's sklearn flavor.
cloudpickle==3.1.1
